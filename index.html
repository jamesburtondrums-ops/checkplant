Setup Guide
  1. Create a Google Sheet (any name, any tab name — this script creates its own tabs).
  2. Extensions → Apps Script. Delete the default code and paste in the script below.
  3. Deploy → New deployment → Web app. Execute as: Me. Who has access: Anyone. Deploy.
  4. Copy the /exec URL into the box above, Save, then Send Test Row.
  5. Watch the Sync Log — it tries POST first, then automatically retries with GET if POST is blocked by your browser.
function doGet(e) {
  if (e && e.parameter && e.parameter.record) {
    try {
      var data = JSON.parse(e.parameter.record);
      return handleRecord(data);
    } catch (err) {
      return jsonResponse({ status: 'error', message: 'Bad record parameter: ' + err.message });
    }
  }
  return ContentService.createTextOutput('CheckPlant receiver is live. Use POST (or GET with ?record=) to submit records.')
    .setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({ status: 'error', message: 'No data received' });
    }
    var d = JSON.parse(e.postData.contents);
    return handleRecord(d);
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.message });
  }
}

function handleRecord(d) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var isService = d.type === 'service';
  var sheetName = isService ? 'Service Records' : 'Check Records';
  var headers = isService
    ? ['Timestamp','Machine','Technician','Service Date','Items (Serviceable/Replaced)','Notes','Signed By','Sign Date']
    : ['Timestamp','Category / Machine ID','Recorded By','Check Date','Items (Pass/Defect)','Notes','Signed By','Sign Date'];

  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) { sheet = ss.insertSheet(sheetName); }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
    sheet.setFrozenRows(1);
  }

  var itemsSummary = '';
  if (d.items && d.items.length) {
    itemsSummary = d.items.map(function (it) { return it.name + ': ' + it.status; }).join(' | ');
  }

  sheet.appendRow([
    new Date(), d.subject || '', d.technician || d.signName || '', d.date || '',
    itemsSummary, d.notes || '', d.signName || '', d.signDate || ''
  ]);

  return jsonResponse({ status: 'ok', sheet: sheetName });
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
